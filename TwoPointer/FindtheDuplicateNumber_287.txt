class Solution {
    
     public int findDuplicate(int[] nums) {
     
         
         // / find the loop first
         // slow and fast
         int slow = 0;
         int fast = 0;
  // did not ceck the condition first since initally yhey both will be at some starting point       
         do{
             slow = nums[slow];
             fast = nums[nums[fast]];
         }while(slow != fast);
         
         
         // once we find the intersection point. reset slow to 0 and move again this will find the fast
         slow = 0;
         
         while(slow != fast){
             slow = nums[slow];
             fast = nums[fast];
         }
         
         return slow;
         
     }
}
